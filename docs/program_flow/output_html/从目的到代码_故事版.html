两条路 / 关系学生

先把画面变成特征，再学习两辆车之间的关系

V-JEPA 在这里负责把 RGB 视频窗口编码成特征。所谓冻结，就是训练下游学生时不改这个编码器的权重。学生接收带空间位置的特征，在画面网格上处理，再汇集两个参与者各自的表示与场景信息。

代码里的 slot_mask 给两个参与者分配网格权重；dense_physics 预测局部存在、深度、速度等辅助量。随后把两个参与者、它们的差、乘积和场景表示拼起来，交给 regressor。输出键名叫 contact，含义仍须结合训练目标协议理解，不能只看到字段名就当成经过校准的事故概率。

CARLA 的几何和运动可以用来生成训练监督。推理时学生读的是视觉特征，教师字段不应变成预测捷径。已有合成小验收验证了真实优化、保存和推理，以及去除教师字段后预测不变；这没有复验真实 CARLA 全量训练，也没有重跑 V-JEPA 官方权重的 RGB 提取。

    def forward(self, tokens: torch.Tensor) -> dict[str, torch.Tensor]:
        batch = tokens.shape[0]
        local = self.project(tokens).transpose(1, 2).reshape(
            batch, -1, self.grid_size, self.grid_size
        )
        local = self.spatial(local).flatten(2).transpose(1, 2)
        mask_logits = self.slot_mask(local).transpose(1, 2)
        dense = self.dense_physics(local)
        weights = torch.sigmoid(mask_logits)
        weights = weights / weights.sum(dim=2, keepdim=True).clamp_min(1e-4)
        slots = torch.einsum("bsp,bph->bsh", weights, local)
        slot_xy = torch.einsum("bsp,pc->bsc", weights, self.patch_xy)
        predicted_presence = torch.sigmoid(dense[:, :, 0])
        predicted_depth = F.softplus(dense[:, :, 1])
        predicted_speed = F.softplus(dense[:, :, 2])
        slot_depth = torch.einsum("bsp,bp->bs", weights, predicted_depth).unsqueeze(-1)
        slot_speed = torch.einsum("bsp,bp->bs", weights, predicted_speed).unsqueeze(-1)
        slot_stats = torch.cat((slot_xy, slot_depth, slot_speed), dim=2).flatten(1)
        first, second = slots[:, 0], slots[:, 1]
        scene_mean = local.mean(dim=1)
        scene_max = local.amax(dim=1)
        relation = torch.cat((
            first, second, torch.abs(first - second), first * second,
            scene_mean, scene_max, slot_stats,
        ), dim=1)
        return {
            "contact": self.regressor(relation),
            "slot_mask_logits": mask_logits,
            "dense": dense,
            "predicted_presence": predicted_presence,
            "relation_embedding": relation,
        }

def project(xyz, pose, row):
    q, t, f, c = camera(row)
    cp = (xyz @ rot(pose).T + pose[:3] - t) @ np.linalg.inv(q.T)
    return (cp[:, :2] / cp[:, 2, None] * f + c, cp[:, 2])

def pose_row(row, pose, origin):
    S = np.diag([1.0, -1.0, 1.0])
    R = rot(pose)
    T = np.eye(4)
    T[:3, :3] = S @ R @ S
    T[:3, 3] = S @ (np.array(pose[:3]) - R @ origin)
    return {**{k: row[k] for k in ['image_height', 'image_width', 'intrinsics', 'camera_matrix']}, 'actor_matrix': T.tolist()}

def exterior_residual(problem, builder, anchors, state, x):
    values = []
    pose = problem.geo.full(x)
    for anchor, association in zip(anchors, state['associations']):
        if association['status'] == 'missing':
            values.append(np.sqrt(anchor['weight'] * 16 / 3))
            continue
        xy, _ = builder.module.project(np.array([association['local']]), pose, problem.row)
        raw = float((xy[0] - anchor['observed_xy']) @ anchor['normal'])

我们把车放回共同的三维坐标后，才有条件比较两辆车各自占据的空间。车中心相距多远不能代替表面间隙：车有长度、宽度和朝向，中心不同样远也可能侧擦或横穿接触。两车接触不要求中心延长线共线。

def box_contact(a, b, tolerance_m=1e-7):
    center_a=np.asarray(a['center'],dtype=float)
    center_b=np.asarray(b['center'],dtype=float)
    axes_a=np.asarray(a['rotation'],dtype=float)
    axes_b=np.asarray(b['rotation'],dtype=float)
    half_a=np.asarray(a['dims'],dtype=float)/2
    half_b=np.asarray(b['dims'],dtype=float)/2
    if not (np.isfinite(center_a).all() and np.isfinite(center_b).all()
            and np.isfinite(axes_a).all() and np.isfinite(axes_b).all()
            and (half_a>0).all() and (half_b>0).all()):
        raise ValueError('Finite centers, rotations, and positive dimensions are required')
    candidates=[axes_a[:,i] for i in range(3)]+[axes_b[:,i] for i in range(3)]
    candidates += [np.cross(axes_a[:,i],axes_b[:,j]) for i,j in itertools.product(range(3),repeat=2)]
    signed_margins=[]
    displacement=center_b-center_a
    for axis in candidates:
        norm=np.linalg.norm(axis)
        if norm<1e-10:continue
        axis=axis/norm
        radius_a=np.sum(half_a*np.abs(axes_a.T@axis))
        radius_b=np.sum(half_b*np.abs(axes_b.T@axis))
        signed_margins.append(float(radius_a+radius_b-abs(displacement@axis)))
    margin=min(signed_margins)
    return dict(overlap=margin>=-tolerance_m,minimum_axis_margin_m=margin,
                meaning='Positive: every tested axis overlaps; negative: a separating axis exists. This margin is not necessarily Euclidean surface distance.')
